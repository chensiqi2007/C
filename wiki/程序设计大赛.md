	#include<stdio.h>  
	#define N 3  
	  
	struct temp_info {  
	    char day [10];  
	    float temp;  
	};  
	  
	int main (void) {  
	    struct temp_info weekWeather[N];  
	    for (int i = 0; i < N; i++) {  
	        printf("Enter the day and temperature for day %d: ", i+1);  
	        scanf("%s %f", weekWeather[i].day, &weekWeather[i].temp);  
	    }  
	    for (int i = 0; i < N; i++) {  
		    printf("Day: %-10s, Temperature: %.6f\n", weekWeather[i].day, weekWeather[i].temp);  
	    }  
	    for (int i=0; i < N-1; i++) {  
	        int index = i;  
        for (int j = i+1; j < N; j++) {  
            if (weekWeather[j].temp > weekWeather[index].temp) {  
                index = j;  
            }  
        }  
        if (i!=index) {  
            struct temp_info temp = weekWeather[i];  
            weekWeather[i] = weekWeather[index];  
            weekWeather[index] = temp;  
        }  
    }  
    printf("\n==========\n");  
    for (int i = 0; i < N; i++) {  
        printf("Day: %-10s, Temperature: %.6f\n", weekWeather[i].day, weekWeather[i].temp);  
    }  
	  
    //保存到文件  
    FILE *fp = fopen("weather.txt", "w");  
    if (fp == NULL) {  
        printf("Error: can not open file\n");  
        return 0;  
    }  
    for (int i = 0; i < N; i++) {  
        fprintf(fp, "%s %.6f\n", weekWeather[i].day, weekWeather[i].temp);  
    }  
    fclose(fp);  
    printf("OK");  
    return 0;  
	}
