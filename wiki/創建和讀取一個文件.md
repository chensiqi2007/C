	#include <commdlg.h>
	#include <stdio.h>

	int main() {
    system("chcp 65001");
    OPENFILENAME ofn;
    char szFile[260] = {0};

    ZeroMemory(&ofn, sizeof(ofn));
    ofn.lStructSize = sizeof(ofn);
    ofn.hwndOwner = NULL;   // 控制台应用中可以设为 NULL
    ofn.lpstrFile = szFile;
    ofn.nMaxFile = sizeof(szFile);
    ofn.lpstrFilter = "Text Files (*.txt)\0*.txt\0All Files (*.*)\0*.*\0";
    ofn.nFilterIndex = 1;
    ofn.lpstrDefExt = "txt"; // 如果用户不加后缀，自动补 .txt
    ofn.Flags = OFN_PATHMUSTEXIST | OFN_OVERWRITEPROMPT;

    if (GetSaveFileName(&ofn)) {
        FILE *fp = fopen(szFile, "w");
        if (fp == NULL) {
            printf("文件创建失败！\n");
            return 1;
        }
        char text[256];
        printf("请输入要写入的内容：");
        scanf("%255s", text);
        fclose(fp);
        printf("文件保存成功：%s\n", szFile);
    } else {
        printf("用户取消选择或出错。\n");
    }

    return 0;
	}

	#include <windows.h>

	int write_to_file_windows (char s[]) {
    	OPENFILENAME ofn;
    	char szFile[260] = {0};

    ZeroMemory(&ofn, sizeof(ofn));
    ofn.lStructSize = sizeof(ofn);
    ofn.hwndOwner = NULL;   // 控制台应用中可以设为 NULL
    ofn.lpstrFile = szFile;
    ofn.nMaxFile = sizeof(szFile);
    ofn.lpstrFilter = "Text Files (*.txt)\0*.txt\0All Files (*.*)\0*.*\0";
    ofn.nFilterIndex = 1;
    ofn.lpstrDefExt = "txt"; // 如果用户不加后缀，自动补 .txt
    ofn.Flags = OFN_PATHMUSTEXIST | OFN_OVERWRITEPROMPT;
    if (GetSaveFileNameA(&ofn)) {
        FILE *fp = fopen(szFile, "w");
        if (fp == NULL) {
            (void)printf("文件创建失败！\n");
            return 1;
        }
        char text[256];
        (void)printf("请输入要写入的内容：");

	}

	int read_from_file_windows (char s[]) {
    OPENFILENAME ofn;
    char szFile[MAX_PATH] = {0};
    HANDLE hFile;
    DWORD dwRead;
    char buffer[1024];
    // 初始化 OPENFILENAME
    ZeroMemory(&ofn, sizeof(ofn));
    ofn.lStructSize = sizeof(ofn);
    ofn.lpstrFile = szFile;
    ofn.nMaxFile = sizeof(szFile);
    ofn.lpstrFilter = "Text Files (*.txt)\0*.txt\0All Files (*.*)\0*.*\0";
    ofn.nFilterIndex = 1;
    ofn.Flags = OFN_PATHMUSTEXIST | OFN_FILEMUSTEXIST;
    if (GetOpenFileName(&ofn)) {
        // 使用 Windows API 打开文件
        hFile = CreateFile(
            szFile,                // 文件路径
            GENERIC_READ,          // 读取权限
            FILE_SHARE_READ,       // 允许共享读取
            NULL,
            OPEN_EXISTING,
            FILE_ATTRIBUTE_NORMAL,
            NULL
        );
        if (hFile == INVALID_HANDLE_VALUE) {
            printf("打開文件失敗，請檢查文件格式！\n");
            return 1;
        }
        printf("文件内容：\n\n");
        // 循环读取文件内容
        while (ReadFile(hFile, buffer, sizeof(buffer) - 1, &dwRead, NULL) && dwRead > 0) {
            buffer[dwRead] = '\0'; // 字符串结尾
            printf("%s", buffer);
        }
        (void)CloseHandle(hFile);
    } else {
        (void)printf("用戶取消或發生未知錯誤\n");
    }
    return 0;
	}

	int main() {
    (void)system("chcp 65001");
    char s[256];
    int select = 0;
    (void)printf("1. 創建文本文件\n2. 讀取文本\n3. 輸入文本到文件(Windows api)\n4. 讀取文件(Windows api)\n");
    (void)printf("請選擇：");
    (void)scanf("%d", &select);
    switch (select) {
        case 1:
            (void)write_to_file(s);
            (void)printf("創建成功\n");
            break;
        case 2:
            (void)read_from_file(s);
            (void)printf("%s",s);
            break;
        case 3:
            (void)write_to_file_windows(s);
            break;
        case 4:
            (void)read_from_file_windows(s);
            break;
        default: ;
    }
	}
