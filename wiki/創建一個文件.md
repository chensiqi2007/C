# 創造一個文件
## 使用C語言代碼，在本地創建一個 .txt 文件

### 代碼
    #include<stdio.h>  
    int main(void) {  
	      char s[10] = "TOM";  
	      FILE *fp = fopen("D:\\a.txt","w");  
	     if (fp==NULL) {  
	        printf("OPEN FILE ERROR!");  
	        return 0;  
	    }  
	    fprintf(fp,"%s",s);  
	    fclose(fp);  
	    printf("OK");  
	    return 0;  
	}

# 讀取一個文件
### 代碼
	#include<stdio.h>  
	int main(void) {  
    char ss[10];  
    FILE *fp = fopen("D:\\a.txt","r");  
    if (fp==NULL) {  
        (void)printf("READ FILE ERROR!");  
        return 0;  
    }  
    (void)fscanf(fp,"%s",ss);  
    (void)fclose(fp);  
    (void)printf("%s",ss);  
    return 0;  
	}
