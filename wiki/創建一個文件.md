# 創造一個文件
## 使用C語言代碼，在本地創建一個 .txt 文件

### 代碼
    #include<stdio.h>  
    int main(void) {  
	      char s[10] = "TOM";  
	      FILE *fp = fopen("D:\\a.txt","w");  
	     if (fp==NULL) {  
	        printf("OPEN FILE ERROR!");  
	        return 0;  
	    }  
	    fprintf(fp,"%s",s);  
	    fclose(fp);  
	    printf("OK");  
	    return 0;  
	}

# 讀取一個文件
### 代碼
	#include<stdio.h>  
	int main(void) {  
    char ss[10];  
    FILE *fp = fopen("D:\\a.txt","r");  
    if (fp==NULL) {  
        (void)printf("READ FILE ERROR!");  
        return 0;  
    }  
    (void)fscanf(fp,"%s",ss);  
    (void)fclose(fp);  
    (void)printf("%s",ss);  
    return 0;  
	}

# 進階
## 使用Windows庫使用戶自行選擇保存位置和文件名稱
### 代碼
	#include <windows.h>
	#include <commdlg.h>
	#include <stdio.h>

	int main() {
    system("chcp 65001");
    OPENFILENAME ofn;
    char szFile[260] = {0};

    ZeroMemory(&ofn, sizeof(ofn));
    ofn.lStructSize = sizeof(ofn);
    ofn.hwndOwner = NULL;   // 控制台应用中可以设为 NULL
    ofn.lpstrFile = szFile;
    ofn.nMaxFile = sizeof(szFile);
    ofn.lpstrFilter = "Text Files (*.txt)\0*.txt\0All Files (*.*)\0*.*\0";
    ofn.nFilterIndex = 1;
    ofn.lpstrDefExt = "txt"; // 如果用户不加后缀，自动补 .txt
    ofn.Flags = OFN_PATHMUSTEXIST | OFN_OVERWRITEPROMPT;

    if (GetSaveFileName(&ofn)) {
        FILE *fp = fopen(szFile, "w");
        if (fp == NULL) {
            printf("文件创建失败！\n");
            return 1;
        }
        char text[256];
        printf("请输入要写入的内容：");
        scanf("%255s", text);
        fclose(fp);
        printf("文件保存成功：%s\n", szFile);
    } else {
        printf("用户取消选择或出错。\n");
    }

    return 0;
	}
