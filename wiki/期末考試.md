    #include<stdio.h>
    #define N 3

    int mysum(int len , int a[]) {
        int sum=0;
        for(int i=1;i<=len;i++)
            sum+=a[i];
        return sum;
    }
    
    int main (void) {
        int a[N];
        for(int i=0;i<N;i++) {
            printf("Please input the %dth number: ",i+1);
            scanf("%d",&a[i]);
        }
    
        for (int i=0;i<N;i++) {
            printf("%5d ",a[i]);
        }
        printf("\n");
        printf("The sum of the array is %d\n",mysum(N,a));

    }
