    #include<stdio.h>
    #define N 3
    
    struct  student
    {
        char name[20];
        char id[9];
        float score;
    };
    
    void creat_student_array (struct student a[] , int len)
    {
        for (int i = 0; i < len ; i++)
        {
            (void)printf("請輸入第%d個學生的學號：\n",i+1);
            (void)scanf("%s",a[i].name);
        
            (void)printf("請輸入第%d個學生的姓名：\n",i+1);
            (void)scanf("%s",a[i].id);
            
            (void)printf("請輸入第%d個學生的分數：\n",i+1);
            (void)scanf("%f",&a[i].score);
        }
    }
    
    void print_student_array (struct student a[] , int len)
    {
        for (int i = 0; i < len ; i++)
        {
            (void)printf("%-10s,%-10s,%2f\n",a[i].id,a[i].name,a[i].score);
            (void)printf("\n ==================== \n");
        }
    }
    
    void bubble_student_array (struct student a[] , int len)
    {
        for (int i=0 ; i < len-1 ; i++)
        {
            for (int j=0 ; j < len-1-i ; j++)
            {
                if (a[j].score > a[j+1].score)
                {
                    struct student temp = a[j];
                    a[j] = a[j+1];
                    a[j+1] = temp;
                }
            }
        }
    }
    
    void select_student_array (struct student a[] , int len)
    {
        for (int i=0 ; i < len-1 ; i++)
        {
            int index = i;
            for (int j=i+1 ; j < len ; j++)
            {
                if (a[index].score < a[j].score)
                {
                    index = j;
                }
            }
            if (i!= index)
            {
                struct student temp = a[i];
                a[i] = a[index];
                a[index] = temp;
            }
        }
    }
    
    int main (void)
    {
        struct student a[N];
        creat_student_array(a,N);
        print_student_array(a,N);
        (void)printf("\n 冒泡排序 正序 \n ==================== \n");
        bubble_student_array(a,N);
        print_student_array(a,N);
        (void)printf("\n 選擇排序 倒序 \n==================== \n");
        select_student_array(a,N);
        print_student_array(a,N);
        return 0;
    }

    int write_txt_to_file_with_windows() {
        OPENFILENAME ofn;
        char szFile[260] = {0};
    
        ZeroMemory(&ofn, sizeof(ofn));
        ofn.lStructSize = sizeof(ofn);
        ofn.hwndOwner = NULL;
        ofn.lpstrFile = szFile;
        ofn.nMaxFile = sizeof(szFile);
        ofn.lpstrFilter = "Text Files (*.txt)\0*.txt\0All Files (*.*)\0*.*\0";
        ofn.nFilterIndex = 1;
        ofn.lpstrDefExt = "txt"; // 如果用户不加后缀，自动补 .txt
        ofn.Flags = OFN_PATHMUSTEXIST | OFN_OVERWRITEPROMPT;
    
        if (GetSaveFileName(&ofn)) {
            FILE *fp = fopen(szFile, "w");
            if (fp == NULL) {
                printf("文件创建失败！\n");
                return 1;
            }
            char text[256];
            printf("请输入要写入的内容：");
            scanf("%255s", text);
            fclose(fp);
            printf("文件保存成功：%s\n", szFile);
        } 
        else {
            printf("用户取消选择或出错。\n");
        }
        
        return 0;
    }
