    #include<stdio.h>
    
    int mysum(int n) {
        int sum = 0 , i = 0;
        for (i = 1; i <= n; i++) {
          sum=sum+i;
        }
    return sum;
    }
    
    int main(){
      int n = 0 , y = 0;
      printf("Input data: ");
      scanf("%d", &n);
      y=mysum(n);
      printf("1+2+3+···+%d= %d\n",n, y);
      return 0;
    }
