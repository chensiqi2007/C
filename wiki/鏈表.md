# 鏈表

## 鏈表的構成
### 首元節點
第一個被使用的有效數據
### 頭節點
存放鏈表信息
### 頭指針
指向鏈表的第一個節點的指針
<br>
<font color="darkorenge">每個鏈表必須有頭文件</font>

## 几個常見的鏈表
### 代碼1
    #include<stdio.h>
    #include <stdlib.h>
    #define N 5
    
    struct student {
      char name[10];
      int score;
      struct student *next;  // 修复：正确的结构体名称
    };
    
    int main() {
      struct student *head = (struct student *)malloc(sizeof(struct student));
      if (head == NULL) {
          printf("Memory allocation failed\n");
          return -1;
      }
    
    head->next = NULL;
    struct student *p = head, *q;

    // 输入学生信息
    for (int i = 0; i < N; i++) {
        q = (struct student *)malloc(sizeof(struct student));
        if (q == NULL) {
            printf("Memory allocation failed\n");
            return -1;
        }
        
        printf("请输入第%d个学生的姓名：", i+1);
        scanf("%s", q->name);
        printf("请输入第%d个学生的成绩：", i+1);
        scanf("%d", &(q->score));  // 修复：添加取地址符

        q->next = NULL;
        p->next = q;
        p = q;
    }
    
    // 输出学生信息
    printf("\n学生信息列表：\n");
    p = head->next;
    while (p != NULL) {
        printf("%10s,%5d\n", p->name, p->score);
        p = p->next;
    }
      
    // 释放内存
    p = head;
    while (p != NULL) {
        struct student *temp = p;
        p = p->next;
        free(temp);
    }
    return 0;
    }
