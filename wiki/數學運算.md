## 計算輸入數字的位數
### 代碼
	#include <stdio.h>  
	#include <stdlib.h>  
	  
	int main(void) {  
    int i = 1;  
    long long a;  
	  
    // 设置控制台编码为UTF-8  
    (void)system("chcp 65001");  
	  
    // 提示用户输入数字  
    (void)printf("請輸入數字：");  
    (void)scanf("%lld", &a);  
	  
    // 计算数字的位数  
    long long temp = a;  // 保存原始值用于计算  
    while (temp / 10 != 0) {  
        i++;  
        temp = temp / 10;  
    }  
	  
    // 输出结果  
    (void)printf("輸入的數字位數：%lld\n", i);  
	  
    return 0;  
	}

## 複數的判定
### 代碼
	#include <stdio.h>  
	#include <stdlib.h>  
	  
	int isprime(int n) {  
    if (n<0||n==0||n==1) {  
        printf("參數不合法");  
        return 0;  
    }  
    for (int i=2; i<n; i++) {  
        if (n%i==0) {  
            printf("是復數");  
            return 0;  
        }  
        else {  
            printf("不是複數");  
            return 0;  
        }  
    }  
    return 1;  
	}  
  
	int main() {  
    system("chcp 65001");  
    int n;  
    printf("請輸入數字：");  
    scanf("%d", &n);  
    isprime(n);  
	}
